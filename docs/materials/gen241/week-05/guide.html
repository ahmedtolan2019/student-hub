الـ for والـ while والـ do while

بصوا يا شباب، الـ loop معناها نكرر نفس التعليمات، وفي كل مرة ممكن القيمة اللي بنشتغل عليها تتغير.

لو عايزين نطبع الأرقام دي، مش لازم نكتب cout لكل رقم:

for (int i = 1; i <= 3; i++)
    cout << i << endl;

هيطبع:

1
2
3

اقروا أول سطر بالترتيب ده:
1. نبدأ i بقيمة 1، وده بيحصل مرة واحدة.
2. قبل كل لفة نسأل: هل i لسه أصغر من أو بتساوي 3؟
3. لو الشرط صح ننفذ cout.
4. بعد التنفيذ i++ تزود i، وبعدين نرجع نسأل على الشرط.

طيب نغير الزيادة بس:

for (int i = 2; i <= 6; i += 2)
    cout << i << endl;

هيطبع:

2
4
6

i += 2 معناها i = i + 2. البداية والحد والزيادة كل واحدة فيهم ليها وظيفة. لو الشيت عايز الأعداد الزوجية، الزيادة هنا جزء من الحل.

نغير اتجاه العد:

for (int i = 3; i >= 1; i--)
    cout << i << endl;

هيطبع:

3
2
1

i-- بتقلل العداد. لازم اتجاه الزيادة يمشي مع الشرط، وإلا الحلقة ممكن ما تخلصش.

طيب إمتى while؟ لما الاستمرار مرتبط بشرط، زي إن المستخدم لسه ما دخلش قيمة الإيقاف. الفرق اللي محتاجين نفهمه قبل السؤال الأول هو مكان اختبار الشرط:

int x = 0;
while (x != 0)
    cout << "while ran" << endl;
do
{
    cout << "do ran" << endl;
} while (x != 0);

هيطبع:

do ran

الـ while اختبرت الأول، فما دخلتش. الـ do نفذت الأول، وبعد كده اختبرت. عشان كده do while مناسبة إننا نقرأ رقم مرة على الأقل.

خلي بالكم من ; بعد شرط for أو while. لو كتبتوها بالغلط، ممكن تعملوا حلقة جسمها فاضي. لكن ; اللي بعد while في آخر do while مطلوبة.

ولو فيه أكتر من تعليمة عايزينهم يتكرروا، حطوهم جوه { }. من غير الأقواس، الحلقة بتكرر التعليمة اللي بعدها بس.

العد والجمع والمتوسط مش نفس الحاجة

يا شباب، لما السؤال يطلب عدد القيم ومجموعها، هنحتاج نخزن حاجتين مختلفتين.

العد بيزيد كل ما تقابلنا قيمة. المجموع بيزيد بقيمة الرقم نفسه. نجرب من غير إدخال عشان نشوف الفرق:

int count = 0, sum = 0;
sum += 3; count++;
cout << count << " " << sum << endl;
sum += -1; count++;
cout << count << " " << sum << endl;
sum += 4; count++;
cout << count << " " << sum << endl;

هيطبع:

1 3
2 2
3 6

كل سطر بيعرض count وبعده sum. الرقم السالب زوّد عدد القيم برضه، لكنه قلل المجموع.

طيب المتوسط؟ بنقسم المجموع على عدد القيم. هنا لازم ننتبه لنوع القسمة:

int sum = 7, count = 4;
double wrong = sum / count;
double correct = (double)sum / count;
cout << wrong << endl;
cout << correct << endl;

هيطبع:

1
1.75

ليه wrong طلعت كده مع إنها double؟ لأن القسمة حصلت الأول بين int وint، فالجزء اللي بعد العلامة ضاع قبل التخزين.

(double)sum بتخلي طرف في العملية عشري قبل ما القسمة تحصل. ينفع كمان نستخدم static_cast<double>(sum). الاتنين بيغيروا نوع القيمة المستخدمة في العملية، مش نوع متغير sum نفسه.

وجربوا حالة مفيش فيها أي قيم. عدد القيم ساعتها صفر، ومينفعش نقسم عليه. لازم نختبر العدد قبل حساب المتوسط.

في السؤال الأول الصفر علامة توقف. مش قيمة نحسبها ضمن العدد. ولو حطيناه في الجمع مش هيغير المجموع، لكن عده هيغيّر المتوسط ويخليه غلط.

fixed وsetprecision، واحدة واحدة

بصوا يا شباب، التنسيق بيظبط شكل الرقم وهو بيتطبع. ناخدها واحدة واحدة بأمثلة.

لو عندي:

double x = 12.5;
cout << fixed << setprecision(2) << x << endl;

هيطبع:

12.50

fixed معناها نعرض الرقم بالشكل العشري الثابت. ومعاها setprecision(2) معناها يظهر رقمين بعد العلامة. عشان كده كمل بصفر.

نغير عدد الأرقام بس:

double x = 12.5;
cout << fixed << setprecision(3) << x << endl;

هيطبع:

12.500

طيب لو الرقم فيه أرقام أكتر من اللي طلبناها؟

double x = 12.3456;
cout << fixed << setprecision(2) << x << endl;
cout << setprecision(3) << x << endl;
cout << setprecision(0) << 12.8 << endl;

هيطبع:

12.35
12.346
13

لاحظوا إنه بيقرّب، مش بيشيل باقي الأرقام وخلاص. ولما طلبنا صفر أرقام بعد العلامة، التقريب حصل برضه.

ليه ما كتبناش fixed تاني في كل سطر؟ لأنها بتفضل شغالة على cout. وsetprecision كمان بتفضل بآخر قيمة حددناها لحد ما نغيرها.

طيب لو نسينا fixed من البداية؟ نجرب البرنامج الصغير ده لوحده:

double x = 12.3456;
cout << setprecision(3) << x << endl;
cout << fixed << setprecision(3) << x << endl;

هيطبع:

12.3
12.346

من غير fixed، الـ 3 بتحدد عدد الأرقام المعنوية في الرقم كله في الوضع العادي. مع fixed بقت عدد الأرقام بعد العلامة. دول مش نفس المعنى.

وآخر حاجة، القيمة اللي جوه المتغير ما اتغيرتش بسبب التنسيق:

double x = 12.3456;
cout << fixed << setprecision(2) << x << endl;
cout << setprecision(4) << x << endl;

هيطبع:

12.35
12.3456

احتجنا iomanip عشان setprecision. اختاروا عدد الأرقام حسب السؤال أو شكل الجدول المطلوب، مفيش عدد ثابت نمشي عليه في كل الشيت.

setw والمسافات وشكل الجدول

يا شباب، setw بتحدد أقل عرض للخانة اللي هتتطبع بعدها. ملهاش علاقة بعدد الأرقام بعد العلامة.

هنحط | على الجنبين عشان المسافات تبقى واضحة:

cout << fixed << setprecision(2);
cout << "|" << 12.5 << "|" << endl;
cout << "|" << setw(8) << 12.5 << "|" << endl;
cout << "|" << setw(3) << 12.5 << "|" << endl;

هيطبع:

|12.50|
|   12.50|
|12.50|

12.50 واخدة خمسة أماكن، فالخانة اللي عرضها تمانية كملت الباقي بمسافات على الشمال. ولما طلبنا عرض أصغر من الرقم، الرقم اتطبع كامل. setw مش بتقص الرقم.

طيب عايزين المسافات على اليمين؟

cout << fixed << setprecision(2);
cout << "|" << left << setw(8) << 12.5 << "|" << endl;
cout << "|" << right << setw(8) << 12.5 << "|" << endl;

هيطبع:

|12.50   |
|   12.50|

left بتحاذي على الشمال، وright على اليمين. اختيار المحاذاة بيفضل شغال، لكن setw نفسها بتتطبق على العنصر اللي بعدها وبعدين لازم نطلب العرض تاني:

cout << "|" << setw(4) << 7 << 8 << "|" << endl;
cout << "|" << setw(4) << 7 << setw(4) << 8 << "|" << endl;

هيطبع:

|   78|
|   7   8|

في السطر الأول العرض اتطبق على 7 بس. في التاني طلبناه لكل رقم.

ودي لقطة مهمة في الهرم: لو كل رقم له خانة، لازم setw تبقى قبل كل رقم. ولو السؤال عايز مسافة واحدة بين القيم، نطبع " "، منبدلهاش بـ setw ونعتبرهم نفس الحاجة.

cout << endl بينزل سطر جديد وبيفرغ مخزن الإخراج. أما " " فدي مسافة في نفس السطر. هنستخدم endl بنفس أسلوب أكواد المحاضرة.

التحويل بين الوحدات: أضرب ولا أقسم؟

بصوا يا شباب، قبل كود التحويل نسأل: الرقم اللي معايا بوحدة إيه، وعايز أطلعه بوحدة إيه؟

cout << fixed << setprecision(3);

باقي القسمة وAND وOR وعدّ اللي اتطبع

بصوا يا شباب، % بتجيب باقي القسمة بين أعداد صحيحة. لما الباقي صفر يبقى الرقم بيقبل القسمة من غير كسر.

cout << 12 % 3 << endl;
cout << 13 % 3 << endl;
cout << 14 % 3 << endl;

هيطبع:

0
1
2

== بنقارن بيها. = بنسند بيها قيمة. و!= معناها لا يساوي.

لو عايزين شرطين يحصلوا مع بعض، نستخدم &&. ولو يكفي واحد فيهم، نستخدم ||. نجرب نفس الرقم:

int x = 6;
cout << (x % 3 == 0) << endl;
cout << (x % 4 == 0) << endl;
cout << ((x % 3 == 0) && (x % 4 == 0)) << endl;
cout << ((x % 3 == 0) || (x % 4 == 0)) << endl;

هيطبع:

1
0
0
1

هنا 1 معناها الشرط صح و0 معناها غلط. الرقم حقق شرط القسمة على 3، لكنه محققش شرط القسمة على 4. عشان كده && رفضته و|| قبلته.

ولو المطلوب يقبل القسمة على 3 وما يقبلش على 6، يبقى شرط القسمة على 6 نتيجته لازم ما تكونش صفر.

طب إيه علاقة ده بخمس أرقام في السطر؟ محتاجين عداد لعدد القيم اللي طبعناها فعلًا. رقم المرشح نفسه مش عداد الطباعة.

نجرب فكرة السطر على أرقام صغيرة، ونخلي السطر فيه تلات قيم للتجربة:

int printed = 0;
for (int x = 2; x <= 8; x += 2)
{
    if (printed % 3 != 0) cout << " ";
    cout << x;
    printed++;
    if (printed % 3 == 0) cout << endl;
}
if (printed % 3 != 0) cout << endl;

هيطبع:

2 4 6
8

المسافة بتتحط قبل القيمة لو مش أول قيمة في السطر. بعد الطباعة نزود العداد. ولما السطر يكمل ننزل سطر جديد. السطر الأخير ممكن يكون ناقص، فبنقفله برضه.

في الشيت هنستخدم العدد اللي السؤال طلبه، وهو خمس قيم في السطر.

الزوايا والراديان وexp

يا شباب، في سؤال الزوايا احنا محتاجين نربط معنى القانون بالدالة اللي هنناديها.

في المثلث القائم، sin بتربط المقابل بالوتر، وcos بتربط المجاور بالوتر. لما نقسم sin على cos نحصل على المقابل على المجاور، ودي tan. ولو عكسنا القسمة تبقى cot.

دوال cmath للزوايا بتستقبل radians، مش degrees. التحويل هو:

radians = degrees * PI / 180.0;

PI هنا بنجيبها بـ acos(-1.0). ده بيدينا قيمة مناسبة للحساب، ومش محتاجين نعتمد على M_PI لأنها مش اسم مضمون في C++98 القياسية.

نجرب أكتر من زاوية:

const double PI = acos(-1.0);
cout << fixed << setprecision(4);
for (int degrees = 20; degrees <= 60; degrees += 20)
{
    double rad = degrees * PI / 180.0;
    cout << degrees << " " << tan(rad) << " " << 1.0 / tan(rad) << endl;
}

هيطبع:

20 0.3640 2.7475
40 0.8391 1.1918
60 1.7321 0.5774

كل سطر زاوية وبعدها tan وبعدها cot. لو دخلنا العدد المكتوب بالدرجات في tan على طول، الدالة هتفهمه كراديان، ومش هتدينا نتيجة الزاوية اللي قصدناها.

وعند صفر، tan تساوي صفر، فـ cot تحتاج قسمة على صفر. بنكتب إنها غير معرّفة، منخترعش لها قيمة صفر.

وفي السؤال اللي بعده هنلاقي e أس x. ده اسمه exponential، وبيظهر مثلًا في نماذج النمو والاضمحلال. المطلوب هنا جدول قيم، مش حل نموذج فيزياء.

cout << exp(0.0) << endl;
cout << exp(1.0) << endl;
cout << exp(2.0) << endl;

هيطبع:

1
2.71828
7.38906

exp(x) بتحسب e أس x مباشرة. أما ^ في C++ فمش أس:

int base = 2, power = 3;
cout << (base ^ power) << endl;
cout << pow(2.0, 3.0) << endl;

هيطبع:

1
8

^ عملية على البتات اسمها XOR. لو الأساس عام نستخدم pow، ولو الأساس e نستخدم exp. كل دول في cmath.

char وASCII والحرف اللي شكله رقم

بصوا يا شباب، char بيخزن محرف واحد. لو كتبنا الحرف بين علامتين مفردتين يبقى بنتكلم عن المحرف، مش عدد بنفس الشكل.

نجرب الحرف والرقم اللي بيمثله في ASCII:

char c = 'A';
cout << c << endl;
cout << int(c) << endl;
cout << static_cast<char>(69) << endl;

هيطبع:

A
65
E

cout عرفت تطبع A كحرف لأن نوعه char. لما كتبنا int(c)، طلبنا طباعة كوده كعدد. والعكس حصل لما حولنا 69 لـ char.

طيب الحرف '7'؟

char c = '7';
cout << c << endl;
cout << int(c) << endl;
cout << c - '0' << endl;

هيطبع:

7
55
7

كود الحرف مش قيمته العددية. طرح '0' ينفع لتحويل خانة واحدة ثبتنا إنها من '0' لحد '9'، لأن أكواد الأرقام متتابعة في ASCII. مش طريقة لتحويل نص فيه عدد كامل.

ومن cctype عندنا isupper وislower وisdigit للاختبار، وtolower للتحويل. الاختبارات بترجع صفر لو غلط وقيمة غير صفر لو صح. متفترضوش إن قيمة الصح لازم تساوي 1.

char c = 'A';
cout << (isupper(c) != 0) << endl;
cout << tolower(c) << endl;
char small = tolower(c);
cout << small << endl;

هيطبع:

1
97
a

ليه tolower اتطبعت كرقم الأول؟ لأن نوع رجوع الدالة int. لما خزنا النتيجة في char اتطبعت كحرف.

إحنا في أسئلة الشيت بنتعامل مع محارف ASCII الإنجليزية. char واحد ودوال cctype بالشكل ده مش حل عام لحرف عربي مكتوب بـ UTF-8. ولو بتتعاملوا مع بايتات خارج ASCII، لازم تراعيوا نوع unsigned char قبل تمريرها لدوال cctype.

الـ nested loops وإزاي نقرأ صف في الهرم

يا شباب، nested loops معناها حلقة جوه حلقة. الخارجية تحدد احنا في أي صف، والداخلية تكمّل شغل الصف ده قبل ما نروح للي بعده.

نبدأ بحاجة أصغر من الهرم:

for (int row = 1; row <= 3; row++)
{
    for (int column = 1; column <= row; column++)
        cout << "*";
    cout << endl;
}

هيطبع:

*
**
***

في كل صف column بتبدأ من 1 من جديد. الحد بتاعها هو row، فكل صف بيطبع عدد مختلف. endl خارج الحلقة الداخلية عشان ننزل سطر بعد اكتمال الصف.

طيب لو عايزين أرقام نازلة بدل النجوم؟

for (int row = 1; row <= 3; row++)
{
    for (int k = row; k >= 1; k--)
        cout << setw(3) << k;

فيه خطوتين منفصلتين: نختبر النوع، وبعدها نحول. مينفعش نعتبر مجرد استدعاء tolower دليل إن المدخل كان uppercase.

ليه بنخزن نتيجة tolower في char؟ لأن الدالة بترجع int. التخزين في char بيخلّي cout تعرضه كمحرف.

سؤال 13: تصنيف المحرف

السؤال الأصلي من الشيت

بصوا يا شباب، هنا مش مطلوب تغيير المحرف. المطلوب نحدد هو حرف كبير، ولا صغير، ولا رقم، ولا خارج الأنواع دي.

المقصود بالرقم هنا خانة رقمية واحدة من '0' لحد '9'، لأننا بنقرأ char، مش عددًا كاملًا.

كل إدخال المفروض ياخد تصنيفًا واحدًا بس.

هنقرأ char، وبعدين نعمل سلسلة if / else if / else:
1. isupper للحرف الكبير.
2. islower للصغير لو الاختبار الأول ما نجحش.
3. isdigit للخانة الرقمية لو السابقين ما نجحوش.
4. else لأي حاجة تانية.

سلسلة else if بتقف عند أول فرع صحيح. ده مناسب للتصنيفات المتنافية هنا.

#include <iostream>

    else if (islower(inputChar)) cout << "Lowercase Letter" << endl;
    else if (isdigit(inputChar)) cout << "Number" << endl;
    else cout << "Invalid Input" << endl;

}

سؤال 16: هرم الأرقام

السؤال الأصلي من الشيت

يا شباب، نفهم الهرم صف صف بدل ما نحاول نحفظ شكل الكود كله.

المطلوب نقرأ N من 1 لحد 15، ونطبع N صفوف. في الصف رقم i، الأرقام تنزل من i لحد 1، وبعدين تطلع من 2 لحد i. وفي أول الصف مسافات تخلي الشكل في النص.

2. نطبع N-i خانات فاضية. كل خانة ثلاث مسافات فعلية.
3. نطبع من i لحد 1 بإنقاص العداد.
4. نطبع من 2 لحد i بزيادة العداد.
5. ننزل سطر بعد انتهاء الجزأين.
